\begin{frame}{Pipeline Overview}

\textbf{Goal}: Validate and correct 37,011 compound structures in
ModelSEED against PubChem, and classify all discrepancies.

\medskip
\small
\textbf{Pipeline} (\texttt{Validate\_PubChem\_Structures.py}):
\begin{description}\setlength\itemsep{1pt}
  \item[Phase 0] Self-consistency --- recompute formula/charge
        from InChI, flag SMILES$\leftrightarrow$InChI mismatches
  \item[Phase 1] External ID lookup --- query PubChem via
        ChEBI/KEGG cross-references (highest confidence)
  \item[Phase 2] Name lookup --- batch compound name queries
        with multi-name disambiguation (for unresolved compounds)
  \item[Phase 3] Recovery --- direct InChIKey/InChI lookup in
        PubChem for compounds still unresolved after Phases~1--2
  \item[Phase 4] Classification --- InChIKey block comparison
        + cascading mismatch sub-classification
  \item[Phase 5] Corrections --- validate stereo (count +
        inversion check) and protonation (ChemAxon $\mathrm{p}K_\mathrm{a}$)
  \item[Phase 6] Output --- write corrected structure files,
        canonicalize SMILES, build Unique file using
        \texttt{List\_ModelSEED\_Structures.py} selection logic,
        record structural and formula conflicts
\end{description}

\end{frame}

\begin{frame}{Correction Details: Stereochemistry (77 accepted, 1,254 rejected)}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\small
\textbf{Acceptance criteria} (all must hold):
\begin{enumerate}\setlength\itemsep{2pt}
  \item PubChem defines $\geq$ stereocenters than stored
  \item Shared stereocenters verified via InChI
        \texttt{/t} (tetrahedral) and \texttt{/b}
        (double-bond) layers --- no inversions allowed
  \item PubChem SMILES/InChIKey internally consistent
\end{enumerate}

\medskip
\textbf{pH\,7 normalization} (20 of 77):
\begin{itemize}\setlength\itemsep{1pt}
  \item PubChem stores neutral forms
  \item Before applying, adjust PubChem SMILES to
        ChemAxon target charge at pH\,7
  \item Prevents overwriting curated ionization states
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\small
\textbf{Rejection reasons} (1,254):
\begin{itemize}\setlength\itemsep{2pt}
  \item PubChem has \textbf{fewer} defined stereocenters
        than stored (360)
  \item Stereo \textbf{inversions} at shared centers (135)
  \item MISMATCH sub-type: wrong mapping, tautomer,
        or different compound (759)
\end{itemize}

\medskip
\textbf{Key design choice}: only \textit{add} missing
stereochemistry, never change existing assignments.
Conservative approach avoids overwriting curated data
from literature sources.
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Correction Details: Protonation (49 accepted, 5 flagged)}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\footnotesize
\textbf{5b. PubChem + pKa} (23 corrections):
\begin{itemize}\setlength\itemsep{1pt}
  \item Compound classified as PROTONATION\_DIFF
  \item ChemAxon $\mathrm{p}K_\mathrm{a}$ confirms that
        the stored charge is wrong at pH\,7
  \item Adjusts \textit{stored} SMILES to target charge
        (does \textbf{not} replace with PubChem's neutral form)
\end{itemize}

\smallskip
\textbf{5c. pKa only} (26 corrections):
\begin{itemize}\setlength\itemsep{1pt}
  \item Runs on \textbf{all} 30,333 compounds
  \item Independent of PubChem comparison
  \item If stored charge $\neq$ ChemAxon target at pH\,7
        and no borderline pKa $\to$ correct
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\footnotesize
\textbf{Safety checks} (both paths):
\begin{itemize}\setlength\itemsep{1pt}
  \item Borderline $\mathrm{p}K_\mathrm{a}$ (6.0--8.0)
        $\to$ flagged, not corrected (5 compounds)
  \item Cross-validation: if ChemAxon ionic charge
        is closer to stored than to target $\to$ skip (90)
  \item After adjustment: verify InChIKey connectivity
        unchanged (protonation-only change)
\end{itemize}

\smallskip
\textbf{Design choice}: use ChemAxon $\mathrm{p}K_\mathrm{a}$
as the authority for pH\,7 ionization, not PubChem
(which stores neutral forms inappropriate for metabolic
modeling).
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Unique Structure Selection (\texttt{-{}-rebuild-unique})}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\small
\textbf{Algorithm} (consistent with
\texttt{List\_ModelSEED\_Structures.py}):
\begin{enumerate}\setlength\itemsep{2pt}
  \item Filter curated/ignored aliases
        (Curation/*.txt, 135 entries)
  \item Priority: Charged InChI $>$ Original InChI
        $>$ Charged SMILE $>$ Original SMILE
  \item Formula conflict $\to$ \textbf{exclude} compound
  \item Multi-database consensus: prefer structures
        appearing in $\geq$2 databases
  \item Source priority fallback:
        MetaCyc $>$ KEGG $>$ ChEBI $>$ Rhea
  \item InChIKey connectivity analysis with
        stereochemistry preference
\end{enumerate}
\end{column}

\begin{column}{0.48\textwidth}
\small
\textbf{Updated Unique file statistics}:
\begin{itemize}\setlength\itemsep{2pt}
  \item 36,883 unique compounds (97,607 rows)
  \item 42 compounds newly included: corrections
        resolved formula conflicts
  \item 3 compounds excluded: unresolved formula
        conflicts (cpd02744, cpd03148, cpd19421)
  \item 2,858 structural conflict entries recorded
  \item 418 formula conflict entries recorded
\end{itemize}

\medskip
\textbf{Standalone usage}:\\[2pt]
\texttt{\footnotesize python Validate\_PubChem\_Structures.py
-{}-rebuild-unique}\\[2pt]
\scriptsize Rebuilds Unique file from All file without
rerunning the validation pipeline.
\end{column}
\end{columns}
