\begin{frame}{Goal \& Approach}

\textbf{Goal}: Validate and correct the 30,321 compound structures in
ModelSEED against PubChem.

\bigskip
\textbf{Approach}:
\begin{enumerate}\setlength\itemsep{4pt}
  \item \textbf{Look up} each compound in PubChem via cross-references
        (ChEBI, KEGG), compound names, and direct structure lookups
        (InChIKey, InChI)
  \item \textbf{Classify} discrepancies by comparing InChIKey blocks:\\
        {\small Block\,1 (connectivity) -- Block\,2 (stereo) -- Block\,3 (protonation)}
  \item \textbf{Correct} when validation criteria are met: stereochemistry
        from PubChem (stereo count + inversion check), protonation to pH\,7
        (OpenBabel + $\text{p}K_\text{a}$ check)
  \item \textbf{Preserve} existing ionization states --- do not
        replace with PubChem's neutral forms
\end{enumerate}

\end{frame}

\begin{frame}{Key Finding: InChIKey Recovery}

\textbf{2,228 compounds} were initially classified as mismatches or
not-found via cross-references and name lookups.

\bigskip
Direct InChIKey lookup in PubChem recovered them ---
\textbf{97\% returned exact InChIKey matches}.

\bigskip
\textbf{Interpretation}: for these compounds, the initial discrepancies
were due to \textbf{incorrect cross-reference mappings or name ambiguity},
not incorrect stored structures.

\begin{frame}{Remaining Work}

\begin{enumerate}\setlength\itemsep{8pt}
  \item \textbf{XREF conflicts} (2,919) --- ChEBI and KEGG map to
        different PubChem CIDs; need resolution strategy
  \item \textbf{MISMATCH review} (1,331) --- sub-classified into
        tautomers, wrong mappings, and structurally distinct compounds
        to prioritize manual review
  \item \textbf{NOT\_FOUND} (832) --- compounds without PubChem
        records; mostly rare metabolites or generic class names
  \item \textbf{AMBIGUOUS} (152) --- compound names map to multiple
        PubChem CIDs
\end{enumerate}

\end{frame}

% =====================================================================
\begin{frame}{Limitations}

\begin{itemize}\setlength\itemsep{6pt}
  \item \textbf{Single reference database} --- validation relies on
        PubChem only; cross-checking against ChEBI, HMDB, or MetaCyc
        would provide independent corroboration
  \item \textbf{OpenBabel pH model} --- empirical $\text{p}K_\text{a}$
        estimation; caught the free pyrophosphate error
        ($\text{p}K_{\text{a}4}{=}9.4$), but other edge cases may exist
  \item \textbf{Name-based lookups} --- inherently noisy; multi-tier
        disambiguation (InChIKey $>$ fingerprint similarity) mitigates
        but does not eliminate errors
  \item \textbf{No confidence scores} --- binary accept/reject
        decisions without uncertainty quantification for borderline
        cases
\end{itemize}

\end{frame}
